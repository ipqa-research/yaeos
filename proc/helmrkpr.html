public subroutine HelmRKPR(nco, NDE, NTD, rn, V, T, Ar, ArV, ArTV, ArV2, Arn, ArVn, ArTn, Arn2)

HelmRKPR

Description

Calculate the reduced residual Helmholtz Energy and it’s derivatives with the RKPR EOS

Arguments

Type IntentOptional Attributes Name
integer, intent(in) :: nco

Number of components in the mixture.
integer, intent(in) :: NDE

Compositional derivative order; 2 also computes Arn2.
integer, intent(in) :: NTD

Flag selecting temperature-dependent derivative calculations.
real(kind=pr), intent(in) :: rn(nco)

Component mole amounts used in mixture rules.
real(kind=pr), intent(in) :: V

Total volume [L].
real(kind=pr), intent(in) :: T

Temperature [K].
real(kind=pr), intent(out) :: Ar

Residual Helmholtz energy.
real(kind=pr), intent(out) :: ArV

First volume derivative of residual Helmholtz energy.
real(kind=pr), intent(out) :: ArTV

Mixed volume-temperature derivative of residual Helmholtz energy.
real(kind=pr), intent(out) :: ArV2

Second volume derivative of residual Helmholtz energy.
real(kind=pr), intent(out) :: Arn(nco)

Mole-number gradient of residual Helmholtz energy.
real(kind=pr), intent(out) :: ArVn(nco)

Mixed volume/mole-number derivative of residual Helmholtz energy.
real(kind=pr), intent(out) :: ArTn(nco)

Mixed temperature/mole-number derivative of residual Helmholtz energy.
real(kind=pr), intent(out) :: Arn2(nco,nco)

Mole-number Hessian of residual Helmholtz energy.
